cmake_minimum_required(VERSION 3.24)
project(atomos_wqk_engine VERSION 3.6.1.16 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ATOMOS_CUDA "Build native CUDA texture accelerator" ON)
option(ATOMOS_SPATIAL "Build inherited S2-dependent spatial targets" ON)
enable_testing()
if(ATOMOS_SPATIAL)
set(ATOMOS_S2_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../tmp/s2_comparison/s2geometry" CACHE PATH "Pinned official S2 checkout")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(FETCH_ABSEIL ON CACHE BOOL "" FORCE)
set(S2_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${ATOMOS_S2_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/s2")
add_library(atomos_spatial INTERFACE)
target_include_directories(atomos_spatial INTERFACE include)
target_link_libraries(atomos_spatial INTERFACE s2)
if(MSVC)
  target_compile_options(s2 PRIVATE /fp:strict)
  target_compile_options(atomos_spatial INTERFACE /fp:strict)
  target_compile_definitions(atomos_spatial INTERFACE NOMINMAX _USE_MATH_DEFINES)
else()
  target_compile_options(atomos_spatial INTERFACE -ffp-contract=off)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/spatial_index_tests.cpp")
  add_executable(spatial_index_tests tests/spatial_index_tests.cpp)
  target_link_libraries(spatial_index_tests PRIVATE atomos_spatial)
  enable_testing()
  add_test(NAME spatial_exact_queries COMMAND spatial_index_tests)
endif()
endif()
if(ATOMOS_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
  if(ATOMOS_SPATIAL)
  add_library(atomos_texture STATIC cuda/texture_index.cu src/texture_index.cpp)
  target_include_directories(atomos_texture PUBLIC include)
  target_link_libraries(atomos_texture PUBLIC s2 CUDA::cudart)
  target_compile_definitions(atomos_texture PUBLIC NOMINMAX _USE_MATH_DEFINES)
  set_target_properties(atomos_texture PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(atomos_texture PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(atomos_texture PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/fp:strict>)
  endif()
  add_executable(compare_s2 src/compare_s2.cpp)
  target_link_libraries(compare_s2 PRIVATE atomos_spatial atomos_texture)
  add_executable(texture_tests tests/texture_tests.cpp)
  target_link_libraries(texture_tests PRIVATE atomos_spatial atomos_texture)
  enable_testing()
  add_test(NAME native_texture_hinges COMMAND texture_tests)
  add_library(atomos_exact STATIC cuda/exact_program.cu)
  target_include_directories(atomos_exact PUBLIC include)
  target_link_libraries(atomos_exact PUBLIC CUDA::cudart)
  set_target_properties(atomos_exact PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/exact_program_tests.cpp")
    add_executable(exact_program_tests tests/exact_program_tests.cpp)
    target_link_libraries(exact_program_tests PRIVATE atomos_exact)
    add_test(NAME native_exact_phi_program COMMAND exact_program_tests)
  endif()
  add_executable(engine_demo src/engine_demo.cpp)
  target_link_libraries(engine_demo PRIVATE atomos_exact atomos_texture atomos_spatial)
  add_test(NAME packed_operator_hinge_integration COMMAND engine_demo)
  add_library(atomos_resident STATIC src/resident_index.cpp cuda/resident_index.cu)
  target_include_directories(atomos_resident PUBLIC include)
  target_link_libraries(atomos_resident PUBLIC s2 CUDA::cudart)
  target_compile_definitions(atomos_resident PUBLIC NOMINMAX _USE_MATH_DEFINES)
  set_target_properties(atomos_resident PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(atomos_resident PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  if(MSVC)
    target_compile_options(atomos_resident PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/fp:strict>)
  endif()
  add_library(atomos_resident_feedback STATIC cuda/resident_feedback.cu)
  target_include_directories(atomos_resident_feedback PUBLIC include)
  target_link_libraries(atomos_resident_feedback PUBLIC atomos_resident CUDA::cudart)
  set_target_properties(atomos_resident_feedback PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(resident_feedback_tests tests/resident_feedback_tests.cpp)
  target_link_libraries(resident_feedback_tests PRIVATE atomos_resident_feedback atomos_spatial)
  add_test(NAME native_resident_feedback COMMAND resident_feedback_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/resident_index_tests.cpp")
    add_executable(resident_index_tests tests/resident_index_tests.cpp)
    target_link_libraries(resident_index_tests PRIVATE atomos_resident atomos_spatial)
    add_test(NAME native_resident_counts COMMAND resident_index_tests)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/compare_resident.cpp")
    add_executable(compare_resident src/compare_resident.cpp)
    target_link_libraries(compare_resident PRIVATE atomos_resident atomos_spatial)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cuda/word_cache_benchmark.cu")
    add_executable(word_cache_benchmark cuda/word_cache_benchmark.cu)
    target_link_libraries(word_cache_benchmark PRIVATE CUDA::cudart)
    set_target_properties(word_cache_benchmark PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  endif()
  endif()
  add_library(tomagi_reference STATIC vendor/wqk_0_6/src/c/tomagi.c)
  target_include_directories(tomagi_reference PUBLIC vendor/wqk_0_6/src/c)
  set_target_properties(tomagi_reference PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  add_library(atomos_tomagi STATIC cuda/tomagi_vm.cu cuda/tomagi_feedback.cu)
  target_include_directories(atomos_tomagi PUBLIC include)
  target_link_libraries(atomos_tomagi PUBLIC CUDA::cudart)
  set_target_properties(atomos_tomagi PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(tomagi_vm_tests tests/tomagi_vm_tests.cpp)
  target_link_libraries(tomagi_vm_tests PRIVATE atomos_tomagi tomagi_reference)
  add_test(NAME native_tomagi_word_machine COMMAND tomagi_vm_tests)
  add_executable(wqk_run tools/wqk_run.cpp)
  target_link_libraries(wqk_run PRIVATE atomos_tomagi CUDA::cudart)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/tomagi_feedback_tests.cpp")
    add_executable(tomagi_feedback_tests tests/tomagi_feedback_tests.cpp)
    target_link_libraries(tomagi_feedback_tests PRIVATE atomos_tomagi tomagi_reference)
    add_test(NAME native_tomagi_word_feedback COMMAND tomagi_feedback_tests "${CMAKE_CURRENT_SOURCE_DIR}/examples/wqk_feedback.tmg" "${CMAKE_CURRENT_SOURCE_DIR}/examples/wqk_feedback.expected.bin")
  endif()
endif()
