cmake_minimum_required(VERSION 3.24)
project(atomOS_COUPLED VERSION 3.6.1.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SATNAV_ENABLE_CUDA "Build the actual CUDA backend" OFF)
option(SATNAV_SANITIZE "Address and undefined-behaviour CPU instrumentation" OFF)
add_library(satnav_core STATIC src/cpu.cpp)
target_include_directories(satnav_core PUBLIC include)
if(SATNAV_ENABLE_CUDA)
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "The requested SM120 profile needs CUDA 12.8 or newer")
  endif()
  find_package(CUDAToolkit REQUIRED)
  target_sources(satnav_core PRIVATE cuda/backend.cu)
  target_compile_definitions(satnav_core PUBLIC SATNAV_HAS_CUDA=1)
  set_target_properties(satnav_core PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(satnav_core PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
  target_link_libraries(satnav_core PUBLIC CUDA::cudart)
endif()
if(MSVC)
 target_compile_options(satnav_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
else()
 target_compile_options(satnav_core PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-ffp-contract=off>)
endif()
add_executable(satnav src/main.cpp)
target_link_libraries(satnav PRIVATE satnav_core)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE satnav_core)
add_executable(self_reference src/self_reference.cpp)
target_link_libraries(self_reference PRIVATE satnav_core)
add_executable(self_reference_tests tests/self_reference_tests.cpp)
target_link_libraries(self_reference_tests PRIVATE satnav_core)
add_executable(coupled_kernel src/coupled_kernel.cpp)
target_link_libraries(coupled_kernel PRIVATE satnav_core)
add_executable(coupled_tests tests/coupled_tests.cpp)
target_link_libraries(coupled_tests PRIVATE satnav_core)
if(SATNAV_ENABLE_CUDA)
 target_sources(self_reference PRIVATE cuda/self_reference.cu)
 set_target_properties(self_reference PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 target_compile_options(self_reference PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
 target_sources(coupled_kernel PRIVATE cuda/coupled_kernel.cu)
 set_target_properties(coupled_kernel PROPERTIES CUDA_ARCHITECTURES "120-real;120-virtual" CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 target_compile_options(coupled_kernel PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
if(MSVC)
 foreach(target self_reference self_reference_tests coupled_kernel coupled_tests)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
 endforeach()
else()
 foreach(target self_reference self_reference_tests coupled_kernel coupled_tests)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
 endforeach()
endif()
if(SATNAV_SANITIZE AND NOT MSVC)
 foreach(target satnav_core satnav core_tests self_reference self_reference_tests coupled_kernel coupled_tests)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined)
 endforeach()
endif()
enable_testing()
add_test(NAME core_reference COMMAND core_tests)
add_test(NAME self_reference_equations COMMAND self_reference_tests)
add_test(NAME coupled_geometry_physics COMMAND coupled_tests)
