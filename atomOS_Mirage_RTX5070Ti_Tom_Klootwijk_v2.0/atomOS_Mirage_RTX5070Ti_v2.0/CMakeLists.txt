cmake_minimum_required(VERSION 3.22)
project(atomOS_Mirage VERSION 2.0 LANGUAGES CXX)
option(ATOMOS_ENABLE_CUDA "Build RTX 5070 Ti CUDA backends" OFF)
option(ATOMOS_SANITIZE "Address/undefined-behavior sanitizers on CPU targets" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(atomos_core STATIC src/runtime.cpp)
target_include_directories(atomos_core PUBLIC include)
add_executable(atomos_cpu src/main.cpp)
target_link_libraries(atomos_cpu PRIVATE atomos_core)
add_executable(atomos_tests tests/core_tests.cpp)
target_link_libraries(atomos_tests PRIVATE atomos_core)
if(MSVC)
  target_compile_options(atomos_core PRIVATE /W4)
else()
  target_compile_options(atomos_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_compile_options(atomos_cpu PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_compile_options(atomos_tests PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(ATOMOS_SANITIZE)
    foreach(t atomos_core atomos_cpu atomos_tests)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
enable_testing()
add_test(NAME core COMMAND atomos_tests)
add_test(NAME propagation COMMAND atomos_cpu --generations 8 --verify --out ${CMAKE_BINARY_DIR}/test_propagation)
add_test(NAME klein COMMAND atomos_cpu --generations 8 --topology klein --verify --out ${CMAKE_BINARY_DIR}/test_klein)
add_test(NAME universal COMMAND atomos_cpu --mode universal --verify --out ${CMAKE_BINARY_DIR}/test_universal)
if(ATOMOS_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "Target CUDA architecture")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "sm_120 requires CUDA Toolkit 12.8 or newer")
  endif()
  find_package(CUDAToolkit 12.8 REQUIRED)
  add_executable(atomos_gpu src/main.cpp cuda/backend.cu cuda/kernels.cu)
  target_include_directories(atomos_gpu PRIVATE include cuda)
  target_compile_definitions(atomos_gpu PRIVATE ATOMOS_GPU_AVAILABLE=1)
  target_link_libraries(atomos_gpu PRIVATE atomos_core CUDA::cudart)
  set_target_properties(atomos_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_SEPARABLE_COMPILATION ON)
  target_compile_options(atomos_gpu PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;-lineinfo>")
  add_test(NAME gpu_texture_propagation COMMAND atomos_gpu --generations 8 --backend texture --verify --out ${CMAKE_BINARY_DIR}/test_gpu_texture)
  add_test(NAME gpu_global_propagation COMMAND atomos_gpu --generations 8 --backend global --verify --out ${CMAKE_BINARY_DIR}/test_gpu_global)
  add_test(NAME gpu_texture_universal COMMAND atomos_gpu --mode universal --backend texture --verify --out ${CMAKE_BINARY_DIR}/test_gpu_vm)
  add_test(NAME gpu_global_universal COMMAND atomos_gpu --mode universal --backend global --verify --out ${CMAKE_BINARY_DIR}/test_gpu_global_vm)
endif()
add_executable(atomos_probe tests/probe.cpp)
target_link_libraries(atomos_probe PRIVATE atomos_core)
if(ATOMOS_SANITIZE AND NOT MSVC)
  target_compile_options(atomos_probe PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(atomos_probe PRIVATE -fsanitize=address,undefined)
endif()
