cmake_minimum_required(VERSION 3.24)
project(atomOS_v3_6_K1 VERSION 3.6.0 LANGUAGES CXX)
option(ATOMOS_ENABLE_CUDA "Build the optional RTX CUDA executable" OFF)
option(ATOMOS_SANITIZE "Address/undefined-behavior sanitizers for CPU review" OFF)
set(CMAKE_CXX_STANDARD 17)

add_executable(atomos_cpu src/main.cpp)

add_executable(atomos_sdf_nor_tests tests/test_sdf_nor.cpp)
add_executable(atomos_sdf_lineage_tests tests/test_sdf_lineage.cpp)
foreach(T atomos_cpu atomos_tests atomos_universal_tests atomos_packed_atlas_tests atomos_klein_tests atomos_sdf_nor_tests atomos_sdf_lineage_tests)
 target_include_directories(${T} PRIVATE include)
 if(MSVC)
  target_compile_options(${T} PRIVATE /W4 /fp:strict)
 else()
  target_compile_options(${T} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(ATOMOS_SANITIZE)
   target_compile_options(${T} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${T} PRIVATE -fsanitize=address,undefined)
  endif()
 endif()
endforeach()
if(ATOMOS_ENABLE_CUDA)

  message(FATAL_ERROR "Use separate CPU sanitizer and CUDA build trees")

 target_link_libraries(atomos_device PUBLIC CUDA::cudart)
 add_executable(atomos_cuda src/main.cpp)
