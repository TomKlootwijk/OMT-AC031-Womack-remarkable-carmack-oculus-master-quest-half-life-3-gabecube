cmake_minimum_required(VERSION 3.24)
project(atomOS_ASA VERSION 3.0.0 LANGUAGES CXX)
option(ASA_ENABLE_CUDA "Build texture/global CUDA paths" OFF)
option(ASA_SANITIZE "Enable host address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(asa INTERFACE)
target_include_directories(asa INTERFACE include)
add_executable(asa_cpu src/main.cpp)
add_executable(asa_tests tests/test_core.cpp)
target_link_libraries(asa_cpu PRIVATE asa)
target_link_libraries(asa_tests PRIVATE asa)
foreach(T asa_cpu asa_tests)
  if(MSVC)
    target_compile_options(${T} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(${T} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    if(ASA_SANITIZE)
      target_compile_options(${T} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${T} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME core_contract COMMAND asa_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME python_contract COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_python.py -v)
  set_tests_properties(python_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
  add_test(NAME cpu_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/integration.py --executable $<TARGET_FILE:asa_cpu>)
endif()
if(ASA_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "120-real;120-virtual")
  endif()
  enable_language(CUDA)
  if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
    message(FATAL_ERROR "sm_120 requires CUDA Toolkit 12.8 or newer")
  endif()
  add_executable(asa_cuda cuda/asa_texture.cu)
  target_link_libraries(asa_cuda PRIVATE asa)
  set_target_properties(asa_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_options(asa_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false> $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  if(Python3_FOUND)
    add_test(NAME gpu_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/integration.py --executable $<TARGET_FILE:asa_cuda> --gpu)
  endif()
endif()
