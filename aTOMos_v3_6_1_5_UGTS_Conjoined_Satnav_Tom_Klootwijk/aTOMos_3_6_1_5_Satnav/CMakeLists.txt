cmake_minimum_required(VERSION 3.24)
project(aTOMosSatnav LANGUAGES CXX)
option(AO_ENABLE_CUDA "Build the CUDA satnav backend" OFF)
option(AO_SANITIZE "CPU Address/UndefinedBehavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(atomos_batch src/batch.cpp)
target_include_directories(atomos_batch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
 target_compile_options(atomos_batch PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4>)
else()
 target_compile_options(atomos_batch PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()
if(AO_ENABLE_CUDA)
 if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES "120-real;120-virtual" CACHE STRING "CUDA targets")
 endif()
 enable_language(CUDA)
 if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
  message(FATAL_ERROR "SM120 requires CUDA Toolkit 12.8 or newer")
 endif()
 find_package(CUDAToolkit REQUIRED)
 target_sources(atomos_batch PRIVATE cuda/backend.cu)
 target_compile_definitions(atomos_batch PUBLIC AO_CUDA=1)
 target_link_libraries(atomos_batch PUBLIC CUDA::cudart)
 set_target_properties(atomos_batch PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
 target_compile_options(atomos_batch PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
add_executable(atomos_satnav src/main.cpp)
target_link_libraries(atomos_satnav PRIVATE atomos_batch)
include(CTest)
if(BUILD_TESTING)
 add_executable(atomos_tests tests/test_core.cpp)
 target_link_libraries(atomos_tests PRIVATE atomos_batch)
 add_test(NAME core COMMAND atomos_tests)
 add_test(NAME help COMMAND atomos_satnav --help)
endif()
if(AO_SANITIZE AND NOT MSVC)
 foreach(t atomos_batch atomos_satnav atomos_tests)
  if(TARGET ${t})
   target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fsanitize=address,undefined;-fno-omit-frame-pointer>)
   target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
 endforeach()
endif()
